import hashlib
import json
from pathlib import Path
import runpy

helpers = runpy.run_path("/source/scripts/prepare-factory-test-tools.py")
old = json.loads(Path("/evidence/prepare.json").read_text())
previous = Path(old["frozen_directory"])
source = Path(old["source_directory"])
evidence = {}
for name, expected_hash in old["binaries"].items():
    old_path = previous / name
    old_hash = helpers["sha256"](old_path)
    source_hash = helpers["sha256"](source / name)
    current_identity = helpers["private_identity"](old_path.stat())
    expected_identity = old["frozen_file_identities"][name]
    assert old_hash == source_hash == expected_hash
    evidence[name] = {
        "original_sha256": expected_hash,
        "current_original_source_sha256": source_hash,
        "current_old_frozen_sha256": old_hash,
        "expected_old_frozen_identity": expected_identity,
        "current_old_frozen_identity": current_identity,
    }
assert any(entry["expected_old_frozen_identity"] != entry["current_old_frozen_identity"] for entry in evidence.values())
print(json.dumps(evidence, indent=2))
PY
